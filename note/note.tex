\section{Définition}
\subsection{Fonction booléenne}
Une fonction booléenne est une fonction qui prend en paramétre un vecteur binaire et qui renvoie 0 ou 1.
$$ f : \mathbb{F}_2^n \to \mathbb{F}_2 $$

Soit $f$ une fonction booléenne en $n$ variable on définie le poid de $f$ comme :
$$ \mathrm{wt}(f) = \#\{f(x) = 1 | x \in \mathbb{F}_2^n\} $$

On peut représenter une fonction par son vecteur d'évaluation (table de vérité) comme :
$$ (f(0), f(1), ..., f(2^n-1)) $$

On définie la distance de hamming de deux fonctions booléenne $f$ et $ g$ comme :
$$ d_H = \#\{f(x) \neq g(x) | x \in \mathbb{F}_2^n\}$$
ou
$$ \mathrm{wt}(f+g) $$

Soit $C$ un Code $\subset \mathbb{F}_2^{2^n}$ et $f$ une fonction booléenne en $n$ variable on définie sa distance au code comme :
$$d(f, C) = \min_{c \in C} d_H(f, c)$$

\subsection{Forme normal algébrique et degrée}
Toute fonctions booléenne peut s'écrire comme une somme de polynôme nommé ANF (algebraic normal form)
$$
f(x)=\sum_{I\subseteq \{1,\dots,n\}} a_I \prod_{i\in I} x_i
$$
On note le degrée d'une fonction comme étant le degrée maximal de ses termes.

\subsection{Code de Reed-Muller}
On définie le code de Reed-Muller en $n$ variable et de degrée $k$ $\mathrm{RM}(k, n)$ comme l'ensemble des fonctions booléennes de degrée inférieur ou égale à k.
$$ \{f \in \mathbb{F}_2^n | deg(f) \leq k\} $$

\subsection{Non linéarité}
Soit un code $C \in \RM{k}{n}$ et une fonction $f \in \FF_2^n$ on définie la non linéarité de degrée $k$ de $f$ comme sa distance au code $C$ autrement dit à sa distance
de toutes les fonctions de degrée au plus k.
$$\mathrm{NL}_k(f) = d(f, C)$$

\subsection{Rayon de recouvrement}
On note $\rho(k, n)$ le rayon de recouvrement d'un code $C \in \mathrm{RM}(k, n)$ la distance maximale entre ce dernier et l'ensemble des fonctions $f \in \FF_2^{2^n}$
$$ \rho(k, n) = \max_{f \in \FF_2^{2^n}} d(f, C) $$

\section{Problématique}
On voudrait connaitre le rayon de recouvrement de $\RM{2}{8}$ et de $\RM{3}{8}$

\subsection{Inégalité de $\rho$}
On sait que
$$ \rho(k, n) \leq \rho(k, n-1) + \rho(k - 1, n - 1) $$
donc pour \RM{2}{8} on voudrait savoir si il existe une fonction $f$ qui atteindrait cette borne supérieure tel que $\mathrm{NL}_2(f) = 96$.
\subsubsection{Décomposition de Plotkin}
Toute fonction booléenne $f \in \FF_2^{2^n}$ peut s'écrire sous la forme
$$f = x_ng + h$$ avec $g \in \RM{k-1}{n-1}$ et $h \in \RM{k}{n-1}$
si $x_n = 0, f = h$ et si $x_n = 1,  f = g + h$ la table de vérité de toute fonction est donc de la forme
$$[h \mid h+g]$$
Ce qui veut dire que pour que $\mathrm{NL}_2(f) = 96$ il faut que pour n'importe quelle fonction $t = x_8v + u \in \RM{2}{8}$
$$\mathrm{wt}(t+f) \geq 96 \equiv \WT{h+v}+\WT{h+v+g+u} \geq 96$$



\section{Algo}
\subsection{Algo probabiliste}
voir Algo \ref{alg:pivotage}

\begin{algorithm}
\caption{Random translate}
\label{alg:pivotage}
\begin{algorithmic}[1]
	\REQUIRE $G$ : Une matrice génératrice d'un code \RM{k}{n} de dimension $[K,N]$
	$f$ : Une fonction booléenne en $n$ variable représenter par sa table de vérité de longueur $N$
\FOR{$i \leftarrow 0$ \TO $K-1$}
    \REPEAT
        \STATE $pivot \leftarrow \text{rand}() \bmod N$
	\UNTIL{$G[i][pivot] = 1$}

    \FOR{$j \leftarrow i$ \TO $K-1$}
	\IF{$j \neq i \AND G[j][pivot] = 1$ }
	    \STATE $G[j] \gets G[j] + G[i]$
        \ENDIF
    \ENDFOR

	\IF {$f[i] = 1$}
		\STATE $f \gets f + G[i]$
	\ENDIF
\ENDFOR
\RETURN $f$
\end{algorithmic}
\end{algorithm}

\subsection{Algo Tervernier like}
Voir Algo \ref{alg:tavernier}

\begin{algorithm}
	\caption{Calcule distance}
	\label{alg:tavernier}
	\begin{algorithmic}[1]
		\REQUIRE Une fonction booléenne en $n$ variable: $f \in \FF_2^{2^n} = x_ng + h$ \newline Le degrée de non linéarité recherché: $k$ \newline
		$N$ et $K$ la longueur et la dimension du code $\RM{k}{n}$
		\STATE best $\gets N$
		\FOR {$u \in \RM{n-1}{k-1}$}
		  \IF {$\WT{g+u} < best$}
		    \FOR {$v \in \RM{k, n-1}$}
		      \STATE best $\gets \min \WT{h+v} + \WT{h+v+g+u}$
		    \ENDFOR
		  \ENDIF
		\ENDFOR
		\RETURN best
	\end{algorithmic}
\end{algorithm}
